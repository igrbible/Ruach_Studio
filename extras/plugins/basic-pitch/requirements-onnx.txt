# `tensorflow<2.15.1` has no wheel. Install basic-pitch itself with --no-deps first:
#
#     pip install --no-deps basic-pitch==0.4.0
#     pip install -r requirements-onnx.txt
#

# present it loads its bundled ONNX model, so the plugin code is unchanged.

typing-extensions
